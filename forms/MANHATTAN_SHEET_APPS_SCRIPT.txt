Shared by BOTH lead-gen widgets (forms/kupot_multistep_exact_v14.html
and forms/form-multi-steps-moria.html) — one Apps Script, one sheet,
matching its existing columns exactly. Frontend-only — no
service-account key involved anywhere, nothing that can leak.

Column mapping used below (both widgets send the same parameter
names, so the same doPost handles either one):
  A Date & Time   → timestamp
  B Name          → lead's name
  C Wa link       → the wa.me link the widget already builds
  D Status        → left blank (matches how existing rows are filled
                    in manually, not by the form)
  E Q1 / F Q2     → left blank (that pair belongs to the older form
                    type already using this sheet — not touched)
  G Form name     → "israel-magazine-kupot-widget" or
                    "israel-magazine-moria-widget" (tells them apart)
  H Source page   → the article page the widget was on
  I Phone number  → lead's phone
  J Q1            → kupot: fund (קופת חולים)          — step 1
                    moria: left blank (not collected by this form)
  K Q2            → kupot: treatment (מה רוצה להסיר)   — step 3
                    moria: lesions (כמה נגעים יש לך)    — step 3
                    (closest available match — moria has no separate
                    "treatment type" question, only lesion count)
  L Q3            → age (קבוצת גיל / קבוצת גיל)         — both forms
  M Q4            → clinic (קליניקה)                    — both forms

function doPost(e) {
  // openById עובד גם אם הסקריפט לא "קשור" ישירות לגיליון (standalone
  // project) — לא תלוי איך פתחתם את העורך.
  const sheet = SpreadsheetApp.openById('17wWkgIqt4xrfTR4efaZSaSjCVwKpVrJHcqbhL9QKOzw').getActiveSheet();
  const data = e.parameter;

  sheet.appendRow([
    data['Timestamp']   || new Date().toLocaleString('he-IL'), // A
    data['Name']        || '',                                  // B
    data['WaLink']       || '',                                 // C
    '',                                                          // D Status
    '',                                                          // E Q1 (legacy)
    '',                                                          // F Q2 (legacy)
    data['FormName']    || '',                                  // G
    data['SourcePage']  || '',                                  // H
    data['Phone']       || '',                                  // I
    data['Fund']        || '',                                  // J Q1
    data['Treatment']   || '',                                  // K Q2
    data['Age']         || '',                                  // L Q3
    data['Clinic']       || ''                                  // M Q4
  ]);

  return ContentService.createTextOutput('OK');
}
